CREATE TABLE MyRep(
	REP_NUM INT AUTO_INCREMENT,
    REP_LNAME VARCHAR(15) NOT NULL,
    REP_FNAME VARCHAR(15) NOT NULL,
    PRIMARY KEY(REP_NUM)
); 

CREATE TABLE Employee(
EMP_NUM INT,
EMP_LNAME VARCHAR(15),
EMP_FNAME VARCHAR(15),
EMP_INITIAL VARCHAR(1),
EMP_HIREDATE DATETIME,
EMP_JOB_CODE INT,
EMP_YEARS INT,
EMP_PROJ_NUM INT 
    
);

CREATE TABLE Job(
	JOB_CODE Int, 
    JOB_DESCRIPTION VARCHAR(25),
    JOB_CHG_HOUR Float,
    JOB_LAST_UPDATE DATETIME
   
);

CREATE TABLE Project(
	PROJ_NUM INT, 
    PROJ_NAME VARCHAR(25),
    PROJ_VALUE Float,
    PROJ_BALANCE Float,
    PROJ_EMP_NUM INT
    
);

CREATE TABLE Assignment(
	ASSIGN_NUM Int(11),
    ASSIGN_DATE DATETIME,
	ASSIGN_PROJ_NUM Int(11),
    ASSIGN_EMP_NUM Int(11),
    ASSIGN_JOB Int(11),
    ASSIGN_CHG_HR Float,
    ASSIGN_HOURS Float,
    ASSIGN_CHARGE Float
    
);
